#!/bin/bash
echo "===================================================="
echo " DVWA + DNS + reCAPTCHA 자동 설치 스크립트"
echo "===================================================="

# 입력값 받기
read -p "사용할 DB 비밀번호 입력: " DBPASS
read -p "사용할 도메인 입력(예: jun.com): " DOMAIN
read -p "서버 IP 입력(예: 172.16.28.170): " SERVERIP
read -p "Google reCAPTCHA 사이트 키 입력: " SITEKEY
read -p "Google reCAPTCHA 시크릿 키 입력: " SECRETKEY

echo "=== 설정 확인 ==="
echo "DB 비밀번호: $DBPASS"
echo "DOMAIN: $DOMAIN"
echo "SERVER IP: $SERVERIP"
echo "SITE KEY: $SITEKEY"
echo "SECRET KEY: $SECRETKEY"
sleep 2

echo "=== 시스템 업데이트 ==="
apt update -y
apt upgrade -y

echo "=== Apache2 + PHP8.4 설치 ==="
apt install apache2 php8.4 php8.4-mysqli php8.4-gd php8.4-xml php8.4-curl php8.4-zip php8.4-mbstring libapache2-mod-php8.4 unzip wget bind9 -y

echo "=== MariaDB 설치 ==="
apt install mariadb-server -y

echo "=== mod_rewrite 활성화 ==="
a2enmod rewrite

echo "=== PHP 설정 패치 ==="
PHPINI="/etc/php/8.4/apache2/php.ini"
sed -i 's/display_errors = Off/display_errors = On/g' $PHPINI
sed -i 's/display_startup_errors = Off/display_startup_errors = On/g' $PHPINI
sed -i 's/allow_url_fopen = Off/allow_url_fopen = On/g' $PHPINI
sed -i 's/allow_url_include = Off/allow_url_include = On/g' $PHPINI

systemctl restart apache2

echo "=== DVWA 다운로드 ==="
cd /var/www/html
rm -rf dvwa
wget https://github.com/digininja/DVWA/archive/refs/heads/master.zip -O dvwa.zip
unzip dvwa.zip
rm dvwa.zip
mv DVWA-master dvwa

echo "=== DVWA config 설정 ==="
cd /var/www/html/dvwa/config
cp config.inc.php.dist config.inc.php

# DB 정보 자동 설정
sed -i "s/'user'.*/'user'] = 'dvwa';/g" config.inc.php
sed -i "s/'password'.*/'password'] = '$DBPASS';/g" config.inc.php
sed -i "s/'database'.*/'database'] = 'dvwa';/g" config.inc.php
sed -i "s/'server'.*/'server'] = '127.0.0.1';/g" config.inc.php

# reCAPTCHA 자동 입력
sed -i "s/'recaptcha_public_key'.*/'recaptcha_public_key'] = '$SITEKEY';/g" config.inc.php
sed -i "s/'recaptcha_private_key'.*/'recaptcha_private_key'] = '$SECRETKEY';/g" config.inc.php

echo "=== MariaDB DVWA DB/계정 생성 ==="
mysql -u root <<EOF
DROP DATABASE IF EXISTS dvwa;
CREATE DATABASE dvwa;
DROP USER IF EXISTS 'dvwa'@'localhost';
CREATE USER 'dvwa'@'localhost' IDENTIFIED BY '$DBPASS';
GRANT ALL PRIVILEGES ON dvwa.* TO 'dvwa'@'localhost';
FLUSH PRIVILEGES;
EOF

echo "=== DVWA 권한 설정 ==="
cd /var/www/html/dvwa
chown -R www-data:www-data /var/www/html/dvwa
chmod -R 777 hackable/uploads/
chmod -R 777 config/

echo "=== AllowOverride 설정 ==="
sed -i 's/<Directory \/var\/www\/>/<Directory \/var\/www\/>\n    AllowOverride All/g' /etc/apache2/apache2.conf

#####################
# DNS 구성
#####################
echo "=== DNS 서버(BIND9) 설정 ==="

cat <<EOF > /etc/bind/named.conf.local
zone "$DOMAIN" {
    type master;
    file "/etc/bind/db.$DOMAIN";
};
EOF

cat <<EOF > /etc/bind/db.$DOMAIN
\$TTL    604800
@       IN      SOA     ns1.$DOMAIN. root.$DOMAIN. (
                        2 604800 86400 2419200 604800 )

@       IN      NS      ns1.$DOMAIN.
ns1     IN      A       $SERVERIP
@       IN      A       $SERVERIP
www     IN      A       $SERVERIP
dvwa    IN      A       $SERVERIP
EOF

systemctl restart bind9

#####################
# Apache VirtualHost 구성
#####################

echo "=== Apache VirtualHost 설정 ==="

VHOST="/etc/apache2/sites-available/dvwa.conf"

cat <<EOF > $VHOST
<VirtualHost *:80>
    ServerName dvwa.$DOMAIN
    DocumentRoot /var/www/html/dvwa

    <Directory /var/www/html/dvwa>
        AllowOverride All
        Require all granted
    </Directory>

    # /dvwa 없이 접속 시 자동 리다이렉트
    RedirectMatch ^/$ /dvwa/login.php
</VirtualHost>
EOF

a2ensite dvwa.conf
systemctl reload apache2

echo "==============================================="
echo " DVWA 설치 완료!"
echo ""
echo "접속 주소:"
echo "   http://dvwa.$DOMAIN"
echo "   http://dvwa.$DOMAIN/setup.php"
echo "==============================================="
