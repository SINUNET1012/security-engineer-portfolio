# 1. 한글 언어팩 설치 (Ubuntu 16.04용)

sudo apt update
sudo apt install language-pack-ko language-pack-ko-base

# 로케일 설정 확인:

locale -a


# ko_KR.utf8 없으면 생성:

sudo locale-gen ko_KR.UTF-8
sudo update-locale LANG=ko_KR.UTF-8

# 2. 한글 폰트 설치

# Ubuntu 16.04는 기본 CJK 폰트가 약해서 글자가 깨질 수 있음.

sudo apt install fonts-nanum ttf-unfonts-core fonts-noto-cjk

# 3. 한글 입력기 설치 (ibus-hangul)
sudo apt install ibus ibus-hangul


# 입력기 활성화:

ibus restart
ibus-setup

# Input Method → Add → Korean → Hangul 선택

# 4. 시스템 언어를 한국어로 변경 (원하면)
# 로케일 파일 수정:
# sudo nano /etc/default/locale
# 아래로 변경:
# LANG=ko_KR.UTF-8
# LANGUAGE=ko_KR:ko
# 저장 후 재부팅:

sudo reboot
