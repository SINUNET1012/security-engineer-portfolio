bs.com.zone

$TTL 1D
@       IN SOA   ns.bs.com. dns.bs.com. (
                                        25123001        ; serial
                                        1D      ; refresh
                                        1H      ; retry
                                        1W      ; expire
                                        3H )    ; minimum
       	NS      ns.bs.com.
        	A       172.16.*.**
	AAAA    ::1
	IN  	MX 10 	mails.bs.com.

ns      	IN      	A      	172.16.*.**
dns	IN	A	172.16.*.**
dbms	IN	A	172.16.*.**

dms	IN	A	172.16.*.**
mails 	IN	A	172.16.*.**
