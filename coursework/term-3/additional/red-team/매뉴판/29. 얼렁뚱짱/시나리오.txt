1. dig ul.com AXFR
# AXFR 결과를 받아 'IN A'가 포함된 라인에서 IP만 뽑아 targets.txt에 저장
1) # IP만 중복 없이 추출
dig @172.16.29.11 ul.com AXFR | grep "IN.A" | awk '{print $5}' | sort -u > targets_ip.txt
2) # [도메인 IP] 형식으로 저장
dig @172.16.29.11 ul.com AXFR | grep "IN.A" | awk '{print $1, $5}' | sed 's/\.$//' > mapping.txt

2. nmap 스캔
# 현재 시간을 파일명에 포함 (예: scan_result_20260204_2330)
FILENAME="scan_result_$(date +%Y%m%d_%H%M)"
nmap -iL targets_ip.txt -sV -p- -oA $FILENAME
