R1
conf t
ipv6 unicast-routing
interface Loopback0
 ip address 1.1.1.1 255.255.255.255
 ipv6 address 2001:1::1/64
 ipv6 eigrp 100
exit
interface FastEthernet0/0
 ip address 10.3.3.2 255.255.255.0
 no shut
exit
interface Serial1/0
 no ip address
 ipv6 address 2002::2/64
 encapsulation frame-relay
 frame-relay map ipv6 2002::3 102 broadcast
 frame-relay map ipv6 2002::4 103 broadcast
 no ipv6 split-horizon eigrp 100
 ipv6 eigrp 100
 no shut
exit
interface Tunnel0
 ipv6 address 2001:14::1/64
 tunnel source 10.3.3.2
 tunnel destination 10.6.6.2
 tunnel mode ipv6ip
 ipv6 eigrp 100
exit
ipv6 router eigrp 100
 eigrp router-id 1.1.1.1
 no shutdown
end
