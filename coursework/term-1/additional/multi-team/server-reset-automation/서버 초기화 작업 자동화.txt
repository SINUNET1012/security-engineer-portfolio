echo "====================================================="
# --- 스크립트 실행 환경 설정 ---
JUPYTER_USER="root"
JUPYTER_BASE_PATH="/root/miniconda3/bin"
JUPYTER_CONFIG_DIR="/root/.jupyter"
JUPYTER_CONFIG_FILE="${JUPYTER_CONFIG_DIR}/jupyter_notebook_config.py"
SERVICE_FILE="/etc/systemd/system/jupyter.service"
NOTEBOOK_PORT=8000

echo "====================================================="
echo "Jupyter Notebook 완전 자동화 스크립트 시작..."
# root 권한 확인
if [[ $EUID -ne 0 ]]; then
    echo "오류: 이 스크립트는 root 권한으로 실행해야 합니다. sudo를 사용해 주세요."
    exit 1
fi

echo "====================================================="
# 1. 네트워크 정보 확인
echo "1. 현재 IP를 확인합니다."
CURRENT_IP=172.16.1.200
echo "현재 시스템 IP: ${CURRENT_IP}"
echo "2. 게이트웨이(172.16.0.1)와 ping 테스트를 실행합니다."
ping -c 5 172.16.0.1
echo "3. DNS 서버 주소 확인."
cat /etc/resolv.conf

echo "====================================================="
# 2. 방화벽 및 SELinux 비활성화
echo "4. 방화벽 및 SELinux 비활성화 시작..."
setenforce 0
sed -i 's/SELINUX=enforcing/SELINUX=disabled/g' /etc/selinux/config
echo "SELinux 설정 완료."
systemctl stop firewalld
systemctl disable firewalld
echo "firewalld 서비스 비활성화 완료."

echo "====================================================="
# 3. Repository 및 패키지 설치
echo "5. Repository 및 필수 패키지 설치..."
dnf install epel-release -y
dnf -y update && dnf -y upgrade
dnf -y install vim wget net-tools

echo "====================================================="
# 4. Miniconda3 설치 (이미 설치되어 있으면 건너뛰기)
echo "6. Miniconda3 설치..."
if [ ! -d "/root/miniconda3" ]; then
    cd ~
    wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh -O miniconda.sh
    chmod +x miniconda.sh
    ./miniconda.sh -b -p /root/miniconda3
    rm miniconda.sh
    /root/miniconda3/bin/conda config --set auto_activate_base false
    echo "export PATH=\$PATH:/root/miniconda3/bin/" >> /etc/bashrc
    source /etc/bashrc
else
    echo "Miniconda3가 이미 설치되어 있습니다. 설치 단계를 건너뜁니다."
fi

echo "====================================================="
# 5. Jupyter Notebook 설치 및 설정
echo "7. Jupyter Notebook 설치 및 초기 설정..."
# 사용자의 요청에 따라 'conda tos accept' 명령어를 유지합니다.
/root/miniconda3/bin/conda tos accept --override-channels --channel https://repo.anaconda.com/pkgs/main
/root/miniconda3/bin/conda tos accept --override-channels --channel https://repo.anaconda.com/pkgs/r
/root/miniconda3/bin/conda install jupyter -y
echo "비밀번호를 설정합니다."
jupyter notebook password


echo "8. jupyter notebook 실행 테스트"
jupyter notebook --generate-config

grep -q "^c.ServerApp.ip" ~/.jupyter/jupyter_notebook_config.py || echo "c.ServerApp.ip = '172.16.1.200'" >> ~/.jupyter/jupyter_notebook_config.py
grep -q "^c.ServerApp.port = 8000" ~/.jupyter/jupyter_notebook_config.py || echo "c.ServerApp.port = 8000" >> ~/.jupyter/jupyter_notebook_config.py
grep -q "^c.ServerApp.root_dir = '/root/'" ~/.jupyter/jupyter_notebook_config.py || echo "c.ServerApp.root_dir = '/root/'" >> ~/.jupyter/jupyter_notebook_config.py
grep -q "^c.ServerApp.open_browser = False" ~/.jupyter/jupyter_notebook_config.py || echo "c.ServerApp.open_browser = False" >> ~/.jupyter/jupyter_notebook_config.py


if [ ! -f "$SERVICE_FILE" ]; then
cat <<EOF | sudo tee "$SERVICE_FILE" > /dev/null
[Unit]
Description=JupyterNotebook
After=network.target

[Service]
User=root
Environment="PATH=/opt/jupyterhub-venv/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/root/miniconda3/bin/"
ExecStart=/root/miniconda3/bin/jupyter notebook --allow-root -f /root/.jupyter/jupyter_notebook_config.py
Restart=always

[Install]
WantedBy=multi-user.target
EOF

  echo "✅ jupyter.service 파일이 생성되었습니다."
else
  echo "⚠️  jupyter.service 파일이 이미 존재합니다. 덮어쓰려면 수동으로 삭제하거나 수정하세요."
fi

echo "====================================================="
# 7. 서비스 활성화 및 시작
systemctl daemon-reload
systemctl start jupyter
systemctl enable jupyter
systemctl restart jupyter

echo "====================================================="
echo "Jupyter Notebook 서비스 설정이 모두 완료되었습니다."
echo "서비스 상태 확인: sudo systemctl status jupyter"
echo "웹 브라우저에서 http://${CURRENT_IP}:${NOTEBOOK_PORT} 로 접속하세요."
