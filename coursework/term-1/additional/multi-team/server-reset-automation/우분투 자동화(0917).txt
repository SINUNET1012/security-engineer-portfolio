#!/bin/bash

#=====================================================
# --- 스크립트 실행 환경 설정 ---
JUPYTER_USER="root"
MINICONDA_FULL_PATH="/root/miniconda3"
JUPYTER_BASE_PATH="${MINICONDA_FULL_PATH}/bin"
JUPYTER_CONFIG_DIR="/root/.jupyter"
JUPYTER_CONFIG_FILE="${JUPYTER_CONFIG_DIR}/jupyter_notebook_config.py"
SERVICE_FILE="/etc/systemd/system/jupyter.service"
NOTEBOOK_PORT=8000

#=====================================================
echo "Jupyter Notebook 자동 설치 스크립트 시작..."

# 1. 시스템 업데이트 및 필수 패키지 설치
echo "1. 시스템 업데이트 및 필수 패키지 설치..."
apt update && apt upgrade -y
apt install -y wget vim net-tools

# 2. Miniconda3 설치
echo "2. Miniconda3 설치 중..."
if [ ! -d "${MINICONDA_FULL_PATH}" ]; then
    wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh -O /tmp/miniconda.sh
    bash /tmp/miniconda.sh -b -p "${MINICONDA_FULL_PATH}"
    rm /tmp/miniconda.sh
    echo "export PATH=\$PATH:${MINICONDA_FULL_PATH}/bin/" >> /etc/bash.bashrc
else
    echo "Miniconda3가 이미 설치되어 있어 설치를 건너뜁니다."
fi

# 3. 환경 변수 적용
source /etc/bash.bashrc

# 4. Jupyter Notebook 설치 및 설정
echo "3. Jupyter Notebook 설치 및 설정..."
${JUPYTER_BASE_PATH}/conda install -y jupyter

if [ ! -f "$JUPYTER_CONFIG_FILE" ]; then
    ${JUPYTER_BASE_PATH}/jupyter notebook --generate-config
fi

# IP, 포트, 토큰 설정 추가 (ens33 인터페이스 기준)
CURRENT_IP=$(ip -4 addr show ens33 | grep -oP '(?<=inet\s)\d+(\.\d+){3}' | head -1)
sed -i "/c.NotebookApp.ip/d" "$JUPYTER_CONFIG_FILE"
echo "c.NotebookApp.ip = '${CURRENT_IP}'" >> "$JUPYTER_CONFIG_FILE"
sed -i "/c.NotebookApp.port/d" "$JUPYTER_CONFIG_FILE"
echo "c.NotebookApp.port = ${NOTEBOOK_PORT}" >> "$JUPYTER_CONFIG_FILE"
sed -i "/c.NotebookApp.token/d" "$JUPYTER_CONFIG_FILE"
echo "c.NotebookApp.token = ''" >> "$JUPYTER_CONFIG_FILE"

---

### 5. 방화벽 비활성화

echo "4. 방화벽(ufw)을 비활성화합니다. 이 작업은 시스템 보안을 낮출 수 있습니다."
ufw disable

---

### 6. Systemd 서비스 파일 생성

echo "5. Systemd 서비스 파일 생성..."
cat << EOF > "$SERVICE_FILE"
[Unit]
Description=Jupyter Notebook Server
After=network.target

[Service]
User=${JUPYTER_USER}
WorkingDirectory=/root
ExecStart=${JUPYTER_BASE_PATH}/jupyter notebook --allow-root --notebook-dir=/root -f "${JUPYTER_CONFIG_FILE}"
Restart=always
[Install]
WantedBy=multi-user.target
EOF

# 7. 서비스 활성화 및 시작
echo "6. 서비스 활성화 및 시작..."
systemctl daemon-reload
systemctl start jupyter
systemctl enable jupyter
systemctl restart jupyter

#=====================================================
echo "Jupyter Notebook 서비스 설정이 완료되었습니다."
echo "웹 브라우저에서 http://${CURRENT_IP}:${NOTEBOOK_PORT} 로 접속하세요."
echo "서비스 상태 확인: sudo systemctl status jupyter"
