import os
import subprocess
import sys
import re
import time


def run_command(command, check=True, input_data=None):
    try:
        process = subprocess.run(command, check=check, input=input_data,
                                 stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
        return process
    except FileNotFoundError:
        print(f"오류: '{command[0]}' 명령어를 찾을 수 없습니다.")
        sys.exit(1)
    except subprocess.CalledProcessError as e:
        print(f"오류: 다음 명령어 실행에 실패했습니다: {' '.join(command)}")
        print(f"에러 메시지: {e.stderr.strip()}")
        sys.exit(1)

def check_root():
    if os.geteuid() != 0:
        print("이 스크립트는 root 권한으로 실행해야 합니다.")
        sys.exit(1)

def get_user_input(prompt):
    full_prompt = f"{prompt.strip()} (종료하려면 'q' 입력): "
    user_input = input(full_prompt)
    if user_input.lower() == 'q':
        print("\n사용자 요청으로 작업을 취소하고 스크립트를 종료합니다.")
        sys.exit(0)
    return user_input

def mdadm_install():
    print("[단계 0] RAID 구성에 필요한 패키지(mdadm, lvm2)를 설치합니다...")
    run_command(["dnf", "install", "-y", "mdadm", "lvm2"])
    print(" - 필수 패키지 설치 완료.")

def find_active_raids():
    if not os.path.exists("/proc/mdstat"):
        return []
    with open("/proc/mdstat", 'r') as f:
        content = f.read()
    active_raids = re.findall(r'^(md\d+)\s*:', content, re.MULTILINE)
    return [f"/dev/{name}" for name in active_raids]

def get_disk_info():
    print("\n시스템 디스크 정보를 스캔 중입니다...")
    used_disks = []
    available_disks = []
    active_raid_members = set()
    if os.path.exists("/proc/mdstat"):
        with open("/proc/mdstat", 'r') as f:
            content = f.read()
            members = re.findall(r'([a-zA-Z0-9]+)\[\d+\]', content)
            for member in members:
                base_disk_name = re.sub(r'p?\d+$', '', member)
                active_raid_members.add(f"/dev/{base_disk_name}")
    result = run_command(["lsblk", "-dno", "NAME,SIZE,TYPE,MOUNTPOINT"])
    for line in result.stdout.strip().split('\n'):
        parts = line.split()
        name, size, disk_type = parts[0], parts[1], parts[2]
        mountpoint = parts[3] if len(parts) > 3 else None
        if disk_type != 'disk' or not (name.startswith('sd') or name.startswith('nvme')):
            continue
        disk_path = f"/dev/{name}"
        if mountpoint and mountpoint != "[SWAP]":
            used_disks.append(f"{disk_path} ({size}) - OS 또는 사용 중")
        elif disk_path in active_raid_members:
            used_disks.append(f"{disk_path} ({size}) - [활성] RAID 구성원")
        else:
            examine_proc = run_command(["mdadm", "--examine", disk_path], check=False)
            if "RAID superblock" in examine_proc.stdout:
                used_disks.append(f"{disk_path} ({size}) - [비활성] RAID 구성원")
            else:
                pv_check_proc = run_command(["pvs", "--noheadings", "-o", "pv_name"], check=False)
                if disk_path in pv_check_proc.stdout:
                    used_disks.append(f"{disk_path} ({size}) - LVM 구성원")
                else:
                    available_disks.append((disk_path, size))
    return sorted(available_disks), sorted(used_disks)

def user_interaction_phase():
    available_disks_with_size, used_disks = get_disk_info()
    
    print("\n--- [디스크 상태 확인] ---")
    if used_disks:
        print("▼ 현재 사용 중인 디스크 (선택 불가):")
        for disk in used_disks: print(f" - {disk}")
    
    if not available_disks_with_size:
        print("\n[오류] RAID 구성에 사용할 수 있는 빈 디스크가 없습니다. 스크립트를 종료합니다.")
        sys.exit(1)
    
    available_disks = [disk[0] for disk in available_disks_with_size]
    
    print("\n▼ RAID 구성에 사용 가능한 디스크:")
    for i, (disk, size) in enumerate(available_disks_with_size):
        print(f"  {i+1}: {disk} ({size})")

    while True:
        try:
            choices_str = get_user_input("\nRAID로 구성할 디스크의 번호를 띄어쓰기로 구분하여 모두 입력하세요 (예: 1 2)")
            choices = choices_str.split()
            selected_indices = [int(c) - 1 for c in choices]
            if all(0 <= i < len(available_disks) for i in selected_indices):
                selected_disks = [available_disks[i] for i in selected_indices]
                break
            else:
                print("오류: 잘못된 번호를 입력했습니다.")
        except ValueError:
            print("오류: 숫자만 입력해주세요.")
            
    raid_levels_info = {'0': 2, '1': 2, '5': 3, '6': 4, '10': 4}
    while True:
        raid_level = get_user_input(f"구성할 RAID 레벨을 입력하세요 ({', '.join(raid_levels_info.keys())})")
        if raid_level not in raid_levels_info:
            print("오류: 지원하지 않는 RAID 레벨입니다.")
            continue
        min_disks = raid_levels_info[raid_level]
        num_disks = len(selected_disks)
        if raid_level == '1' and num_disks != 2:
            print("오류: RAID 1은 반드시 2개의 디스크가 필요합니다.")
        elif raid_level == '10' and num_disks % 2 != 0:
            print("오류: RAID 10은 반드시 짝수 개의 디스크가 필요합니다.")
        elif num_disks < min_disks:
            print(f"오류: RAID {raid_level}은(는) 최소 {min_disks}개의 디스크가 필요합니다. (현재 {num_disks}개 선택)")
        else:
            break
            
    raid_num = 0
    while os.path.exists(f"/dev/md{raid_num}"):
        raid_num += 1
    raid_device = f"/dev/md{raid_num}"
    print(f"\n자동으로 할당된 RAID 장치 이름: {raid_device}")

    default_mount_point = f"/mnt/md{raid_num}_storage"
    user_input = get_user_input(f"마운트할 디렉토리 경로를 입력하세요 (기본값: {default_mount_point})")
    mount_point = user_input if user_input else default_mount_point
    
    fs_type = "xfs"

    print("\n--- [최종 설정 확인] ---")
    print(f"  - 대상 디스크: {', '.join(selected_disks)}")
    print(f"  - RAID 레벨: {raid_level}")
    print(f"  - RAID 장치: {raid_device}")
    print(f"  - 마운트 경로: {mount_point}")
    print(f"  - 파일 시스템: {fs_type}")
    
    return selected_disks, raid_level, raid_device, mount_point, fs_type

def prepare_disks(disks, all_active_raids):
    """
    모든 활성 RAID를 중지하고, 선택된 각 디스크를 순차적으로 완벽하게 정리 및 파티션합니다.
    """
    print("\n[단계 1] 디스크 준비 작업을 시작합니다 (정리 및 파티션)...")
    
    # 1. 시스템의 모든 활성 RAID 어레이를 먼저 중지
    if all_active_raids:
        print(" - 시스템에서 감지된 모든 활성 RAID를 중지합니다...")
        for raid_dev in all_active_raids:
            run_command(["mdadm", "--stop", raid_dev], check=False)
    
    # 2. 선택된 각 디스크를 하나씩, 완벽하게 정리하고 파티션
    for i, disk in enumerate(disks):
        print(f"\n- [{i+1}/{len(disks)}] '{disk}' 디스크 작업 중...")
        
        # 2-1. 모든 메타데이터(LVM, RAID 등)를 물리적으로 삭제
        print(f"   -> {disk} 초기화...")
        run_command(["dd", "if=/dev/zero", f"of={disk}", "bs=1M", "count=100"], check=False)
        run_command(["pvremove", "-ff", "-y", disk], check=False)
        run_command(["mdadm", "--zero-superblock", disk], check=False)
        
        # 2-2. 새 파티션 생성
        print(f"   -> {disk} 파티션 생성...")
        fdisk_script = "o\nn\np\n1\n\n\nt\nfd\nw\n"
        run_command(["fdisk", disk], input_data=fdisk_script)
        
        # 2-3. 변경사항을 즉시 커널에 알림
        print(f"   -> {disk} 변경사항 적용...")
        run_command(["partprobe", disk], check=False)
        run_command(["blockdev", "--rereadpt", disk], check=False)

    print("\n - 모든 디스크 준비 완료. 안정화를 위해 잠시 대기합니다...")
    time.sleep(5) # 모든 디스크 작업 후 충분히 대기

def toggle_lvm_service(action):
    service = "lvm2-monitor.service"
    if action == "stop":
        print(f"\n -> LVM 자동 활성화 서비스를 임시로 중지합니다... ({service})")
        run_command(["systemctl", "stop", service], check=False)
    elif action == "start":
        print(f"\n -> LVM 자동 활성화 서비스를 다시 시작합니다... ({service})")
        run_command(["systemctl", "start", service], check=False)

def get_partition_path(disk_path):
    if "nvme" in disk_path:
        return f"{disk_path}p1"
    else:
        return f"{disk_path}1"

def create_raid_array(raid_device, raid_level, disks):
    print("\n[단계 2] RAID 어레이를 생성합니다...")
    disk_partitions = [f"{disk}1" for disk in disks]
    num_disks = len(disk_partitions)
    run_command(["mdadm", "--create", "--verbose", raid_device, 
                 f"--level={raid_level}", f"--raid-devices={num_disks}", 
                 *disk_partitions])
    print(" - RAID 어레이 생성 완료.")


    def save_mdadm_config():
    print("\n[단계 3] RAID 설정을 시스템에 저장합니다...")
    conf_dir = "/etc/mdadm"
    conf_file = f"{conf_dir}/mdadm.conf"
    if not os.path.exists(conf_dir):
        os.makedirs(conf_dir)
    if not os.path.exists(conf_file):
        open(conf_file, 'a').close()
    result = run_command(["mdadm", "--detail", "--scan"])
    new_line = result.stdout.strip()
    with open(conf_file, 'r+') as f:
        existing_content = f.read()
        if new_line not in existing_content:
            f.write(new_line + "\n")
            print(f" - {conf_file} 파일에 새 RAID 정보 추가 완료.")
        else:
            print(f" - 이미 {conf_file} 파일에 해당 정보가 존재합니다.")

def format_and_mount(raid_device, mount_point, fs_type):
    print(f"\n[단계 4] {raid_device}를 {fs_type}으로 포맷하고 마운트합니다...")
    run_command([f"mkfs.{fs_type}", "-f", raid_device])
    os.makedirs(mount_point, exist_ok=True)
    run_command(["mount", raid_device, mount_point])
    print(" - 포맷 및 마운트 완료.")

def setup_auto_mount(raid_device, mount_point, fs_type):
    print("\n[단계 5] 재부팅 시 자동 마운트를 위해 /etc/fstab을 설정합니다...")
    result = run_command(["blkid", "-s", "UUID", "-o", "value", raid_device])
    uuid = result.stdout.strip()
    fstab_file = "/etc/fstab"
    fstab_line = f"UUID={uuid}\t{mount_point}\t{fs_type}\tdefaults\t0 0\n"
    with open(fstab_file, 'r') as f:
        if any(uuid in line for line in f):
            print(" - 이미 /etc/fstab에 해당 UUID가 등록되어 있습니다.")
            return
    with open(fstab_file, 'a') as f:
        f.write(fstab_line)
    print(" - /etc/fstab 파일에 자동 마운트 설정 추가 완료.")


def main():
    """ 스크립트의 메인 실행 함수 """
    check_root()
    mdadm_install()
    
    all_active_raids = find_active_raids()
    disks, level, device, mount_p, fs = user_interaction_phase()
    
    final_check = get_user_input(f"\n[!!!] 최종 경고: 선택한 디스크({', '.join(disks)})의 모든 데이터가 영구적으로 삭제됩니다.\n정말로 진행하시겠습니까? (yes/no)")
    if final_check.lower() != 'yes':
        print("작업을 취소했습니다.")
        sys.exit(0)
    
    try:
        toggle_lvm_service("stop")
        
        prepare_disks(disks, all_active_raids)
        
        create_raid_array(device, level, disks)
    finally:
        toggle_lvm_service("start")

    save_mdadm_config()
    format_and_mount(device, mount_p, fs)
    setup_auto_mount(device, mount_p, fs)
    
    print("\n" + "="*50)
    print("모든 RAID 설정이 성공적으로 완료되었습니다!")
    print("\n▼ 현재 RAID 상태:")
    subprocess.run(["mdadm", "--detail", device])
    print("\n▼ 현재 마운트 상태:")
    subprocess.run(f"df -h | grep --color=never {mount_p}", shell=True)
    print("="*50)

if __name__ == "__main__":
    main()
