echo "7. Miniconda 설치"
cd ~

expect -c "
set timeout 50
spawn ./Miniconda3-latest-Linux-x86_64.sh
expect \">>>\"
send \"\r\"

expect \">>>\"
send \"yes\r\"

expect \">>>\"
send \"\r\"

expect -re \"\\[no\\].*>>>\"
send \"yes\r\"

expect eof
"
