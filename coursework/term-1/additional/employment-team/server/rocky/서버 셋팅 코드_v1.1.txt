echo "초기화를 시작합니다."
echo "1. 현재 IP를 확인합니다."
ip a | grep inet

echo "2. 게이트웨이와 ping 테스트를 실행합니다."
ping -c 5 172.16.0.1

echo "3. DNS Server 주소 확인."
cat /etc/resolv.conf

echo "4. Repository 설치(EPEL, REMI)"
sudo dnf install epel-release -y
sudo dnf install https://rpms.remirepo.net/enterprise/remi-release-9.rpm -y

echo "5. 패키지 설치"
sudo dnf -y install vim
sudo dnf -y install net-tools
sudo dnf -y install wget
sudo dnf -y install yum
sudo dnf -y install expect

echo "6. 설치된 패키지들 업데이트 및 업그레이드"
sudo dnf -y update && dnf -y upgrade

####miniconda3 설치 expect 자동화(miniconda_installation.sh)

echo "7. Miniconda 설치"
cd ~
wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh
chmod 744 Miniconda3-latest-Linux-x86_64.sh
./Miniconda3-latest-Linux-x86_64.sh

######################

echo "8. 환경변수 등록"
echo "export PATH=$PATH:/root/miniconda3/bin/" >> /etc/bashrc
source /etc/bashrc

conda config --set auto_activate_base false
conda deactivate

echo "9. Jupyternotebook 설치"
conda tos accept --override-channels --channel https://repo.anaconda.com/pkgs/main
conda tos accept --override-channels --channel https://repo.anaconda.com/pkgs/r
conda install jupyter -y

echo "10. 방화벽 해제 및 확인"
sudo sed -i 's/^SELINUX=.*/SELINUX=disabled/' /etc/selinux/config
sudo systemctl disable --now firewalld
sudo systemctl status firewalld --no-pager

echo "11. Jupyternotebook 실행"
IP=`ip -4 -o addr show dev ens160 | awk '{print $4}' | cut -d/ -f1 | head -n1`
jupyter notebook --allow-root --ip=${IP} --port=8000 &
