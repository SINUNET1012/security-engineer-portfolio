#append() 함수란 추가하는 함수로 맨 끝의 인덱스에 원하는 것을 추가할 수 있다.
#여러가지를 추가해도 하나의 요소로 취급한다.

a = [1,2,3,4]
a.append(0)
a.append("1")
a.append([1,2])
print(a)
print(a[6])

결과 

[1, 2, 3, 4, 0, '1', [1, 2]]
[1, 2]


#clear() 함수란 리스트의 모든 요소를 제거하여 빈 리스트로 만든다.
a = [1,2,3,4]
a.clear()
print(a)

결과

[1, 2, 3, 4]
[]

#copy() 함수란 복제하는 함수로 원하는 리스트를 복제할 수 있다. 
#얕은 복사를 하여 중첩 리스트를 완전히 복제하지 않는다.
a = [1,2,3,4,[1,2]]
print(a) 
b = a.copy()
print(b)
a[4][1] = 100
a[1] = 10
print(a)
print(b)

결과

[1, 2, 3, 4, [1, 2]]
[1, 2, 3, 4, [1, 2]]
[1, 10, 3, 4, [1, 100]]
[1, 2, 3, 4, [1, 100]]

#count() 함수란 리스트 안에 특정 요소가 몇 번 나오는지 세어준다.
a = [1,1,1,1,1,1,2,3,4]
print(a.count(1))

결과

6

#extend() 함수는 리스트의 크기를 확장하는 함수이다. append() 함수와 다르게 원하는 만큼 문자열과 숫자를 넣을 수 있다.
a = []
a.extend([0,2,3,4])
print(a)

결과 

[0, 2, 3, 4]

#insert 는 리스트에서 데이터를 삽입하는 함수일 것이라고 추측

print(a)                   #인덱스 1번에 6이 추가되면서 리스트a의 길이도 1 증가한것을 확인할 수 있다.

#pop는 리스트 객체에서 데이터를 빼온다고 추측
a = [1, 2, 3, 4, 5]
b = a.pop(2)
print(a, b)                    #인덱스 2번의 데이터가 반환된 것을 확인할 수 있다.
>
[1, 2, 4, 5] 3

#예제
a = [1, 2, 3, 4, 5, 6]
print("선택할 번호를 고르시오 : %s" %(a))
b = a.pop(int(input("번호 :"))-1)                    #pop함수는 해당하는'인덱스의 위치'에 있는 값을 반환하는 것이다.
print("선택한 번호: %d    남은 번호 :%s" %(b, a))      #선택한 '데이터 값'을 반환하는 것이 아니다. 그래서 b에 -1해준거다.
>
선택할 번호를 고르시오 : [1, 2, 3, 4, 5, 6]
번호 : 3
선택한 번호: 3    남은 번호 :[1, 2, 4, 5, 6]

# reverse는 **자료형(특히 리스트)**을 역순으로 뒤집을 때 사용
# .reverse() 메서드 > (리턴값) : 리스트 자체를 **제자리(in-place)**에서 뒤집음
lst = [1, 2, 3, 4]
lst.reverse()
print(lst)
#원본 리스트가 바뀜
# reversed() 함수 : 이터레이터를 리턴함
lst = [1, 2, 3, 4]
rev = reversed(lst)
print(list(rev))
#원본 리스트는 그대로
[4, 3, 2, 1]
[4, 3, 2, 1]



# sort는 리스트의 원본자체를 정렬
# .sort() 메서드 : 오름차순으로 정렬
a = [5, 7, 2, 4, 9]
a.sort()
print(a)
#원본 리스트가 바뀜
# sorted()는 내장 함수(function)
a = [3, 1, 2, 7, 5, 4, 9]
b = sorted(a)
print(a)
print(b)
#원본 리스트는 그대로
[2, 4, 5, 7, 9]
[3, 1, 2, 7, 5, 4, 9]
[1, 2, 3, 4, 5, 7, 9]
